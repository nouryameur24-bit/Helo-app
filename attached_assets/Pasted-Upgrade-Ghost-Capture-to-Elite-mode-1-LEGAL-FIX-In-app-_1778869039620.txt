Update the ghostCaptureSave function:

export async function ghostCaptureSave(params: {
  barcode: string;
  productName: string;
  category: string;
  ocrText: string;
  verdict: VerdictResult;
  trimester: Phase;
}): Promise<boolean> {
  if (!isSupabaseConfigured) return false;
  try {
    // Check if already captured before
    const { data: existing } = await supabase
      .from('community_submissions')
      .select('id, metadata')
      .eq('barcode', params.barcode)
      .single();

    if (existing) {
      const meta = existing.metadata || {};
      const scanCount = (meta.scan_count || 1) + 1;
      const newStatus = scanCount >= 3 ? 'community_verified' : 'auto_captured';
      await supabase
        .from('community_submissions')
        .update({ 
          status: newStatus,
          metadata: { ...meta, scan_count: scanCount, last_scanned: new Date().toISOString() }
        })
        .eq('id', existing.id);
    } else {
      await supabase.from('community_submissions').insert({
        barcode: params.barcode,
        product_name: params.productName,
        category: params.category,
        ingredients_photo_url: null,
        status: 'auto_captured',
        metadata: {
          ingredients_raw: params.ocrText,
          ai_verdict: params.verdict,
          trimester: params.trimester,
          scan_count: 1,
          first_captured: new Date().toISOString(),
        },
      });
    }
    return true;
  } catch {
    return false;
  }
}
