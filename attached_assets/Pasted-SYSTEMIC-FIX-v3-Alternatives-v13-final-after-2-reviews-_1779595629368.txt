function matchFoodContextOverride(
  ingredientName: string,
): { matched: true; label: string; riskLevel: 'safe' | 'caution' } | null {
  for (const { pattern, label, riskLevel } of FOOD_CONTEXT_OVERRIDES) {
    if (pattern.test(ingredientName)) return { matched: true, label, riskLevel };
  }
  return null;
}

PART 2 — alternatives.ts: Adaptive gate + absolute floor (Map pattern)
═══════════════════════════════════════════════════════════

FILE: artifacts/api-server/src/routes/alternatives.ts

A. Add glow_score to AlternativeProductDto:

  }

B. In the belt loop, BEFORE the loop, declare the score map:

C. REPLACE the belt loop body with:

  for (const b of sniperResult.barcodes) {
    const cand = byBarcode.get(b);
    if (!cand) continue;
    const parsed = parseIngredients(cand.ingredients_raw);
    const belt = await matchDeterministic(parsed, phase, beltDomain);
    if (belt.dangerousMatch) continue;
    
    // ── Compute candidate's GlowScore ────────────────────
    const candidateDangerCount = belt.matches.filter(m => m.riskLevel === 'danger').length;
    const candidateCautionCount = belt.matches.filter(m => m.riskLevel === 'caution').length;
    const candidateUnknownCount = belt.matches.filter(m => !m.matched).length;
    const candidateGlowScore = Math.max(0, Math.min(100,
      100 - (candidateDangerCount * 50) - (candidateCautionCount * 15) - (candidateUnknownCount * 2)
    ));
    
    // ── ABSOLUTE FLOOR: only display safe alternatives ───
    // Never show a "yellow caution" alternative — UX consistency
    if (candidateGlowScore < 80) {
      req.log.info({
        action: 'rejected_below_safe_floor',
        candidateBarcode: b,
        candidateScore: candidateGlowScore,
      }, 'alternative rejected: not safe enough to display');
      continue;
    }
    
    // ── ADAPTIVE GATE: only when origin is problematic ───
    // - Origin score unknown → no gate (don't invent threshold)
    // - Origin already healthy (>= 70) → no gate (diversify by brand instead)
    // - Origin problematic (< 70) → require +10 minimum improvement
    const originGlowScore: number | null = (phaseCache as any)?.glow_score ?? null;
    const requireImprovement = originGlowScore !== null && originGlowScore < 70;
    if (requireImprovement && candidateGlowScore < originGlowScore + 10) {
      req.log.info({
        action: 'rejected_insufficient_improvement',
        candidateBarcode: b,
        candidateScore: candidateGlowScore,
        originScore: originGlowScore,
        threshold: originGlowScore + 10,
      }, 'alternative rejected');
      continue;
    }
    
    // Store computed score for DTO (Map pattern, no mutation)
    scoreByBarcode.set(b, candidateGlowScore);
    
    // ── Existing categoryTag check ────────────────────────
    if (categoryTag && liveStrategy === 'keywords' 
        && !cand.categories_tags.includes(categoryTag)) {
      continue;
    }
    
    // ── IMPROVED junk brand filter ───────────────────────
    const brandNorm = cand.brand.trim().toLowerCase();
    const categoryNorm = cand.category.trim().toLowerCase();
    
    if (brandNorm.length < 2) {
      if (!cand.image_url) continue;
      cand.brand = 'Marque distributeur';
    } else if (brandNorm === categoryNorm) {
      cand.brand = 'Marque distributeur';
    }
    
    safeCandidates.push(cand);
  }

D. In DTO mapping, use the Map:

  const dtos: AlternativeProductDto[] = safeCandidates.map(c => {

    };
  });
