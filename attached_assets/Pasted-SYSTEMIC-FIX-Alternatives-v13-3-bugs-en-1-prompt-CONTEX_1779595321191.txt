// ─── Food Context Overrides ────────────────────────────────
// Common food ingredients that match dangerous dictionary rows
// but are negligible in food context (processing residuals,
// trace amounts, regulated additives).
//
// These overrides apply ONLY when domain === 'food'. They run
// AFTER SAFE_OVERRIDES (which handle compound names like 
// "vinaigre d'alcool") and BEFORE the main matching loop.

const FOOD_CONTEXT_OVERRIDES: { 
  pattern: RegExp; 
  label: string; 
  riskLevel: 'safe' | 'caution';
}[] = [
  // Standalone alcohol = manufacturing residual (<0.5%)
  { pattern: /^alcool$/iu, label: 'Alcool (résidu de fabrication)', riskLevel: 'caution' },
  { pattern: /^alcohol$/iu, label: 'Alcohol (processing residual)', riskLevel: 'caution' },
  { pattern: /^[ée]thanol$/iu, label: 'Éthanol (résidu)', riskLevel: 'caution' },
  { pattern: /\balcool\s+[ée]thylique\b/iu, label: 'Alcool éthylique (résidu)', riskLevel: 'caution' },
  
  // E-numbers safe by default (regulated EU additives)
  { pattern: /\bE\s?322\b/iu, label: 'E322 Lécithine', riskLevel: 'safe' },
  { pattern: /\bE\s?330\b/iu, label: 'E330 Acide citrique', riskLevel: 'safe' },
  { pattern: /\bE\s?300\b/iu, label: 'E300 Vitamine C', riskLevel: 'safe' },
  { pattern: /\bE\s?150a\b/iu, label: 'E150a Caramel simple', riskLevel: 'safe' },
  { pattern: /\bE\s?440\b/iu, label: 'E440 Pectine', riskLevel: 'safe' },
  
  // Common safe food ingredients
  { pattern: /\bl[ée]cithine\s+de\s+soja\b/iu, label: 'Lécithine de soja (émulsifiant)', riskLevel: 'safe' },
  { pattern: /\bl[ée]cithines?\b/iu, label: 'Lécithine', riskLevel: 'safe' },
  { pattern: /\bacide\s+citrique\b/iu, label: 'Acide citrique (E330)', riskLevel: 'safe' },
  { pattern: /\bacide\s+ascorbique\b/iu, label: 'Vitamine C', riskLevel: 'safe' },
  { pattern: /\bcolorant\s+caramel\b/iu, label: 'Colorant caramel', riskLevel: 'safe' },
  { pattern: /\bar[oô]mes?\s+naturel/iu, label: 'Arôme naturel', riskLevel: 'safe' },
  { pattern: /\bgomme\s+(?:xanthane|guar|arabique|caroube)\b/iu, label: 'Gomme épaississante', riskLevel: 'safe' },
];

function matchFoodContextOverride(
  ingredientName: string,
): { matched: true; label: string; riskLevel: 'safe' | 'caution' } | null {
  for (const { pattern, label, riskLevel } of FOOD_CONTEXT_OVERRIDES) {
    if (pattern.test(ingredientName)) return { matched: true, label, riskLevel };
  }
  return null;
}

THEN in matchDeterministic(), AFTER the existing SAFE_OVERRIDES check 
and BEFORE the main matching loop, add:

  for (const b of sniperResult.barcodes) {
    const cand = byBarcode.get(b);
    if (!cand) continue;
    const parsed = parseIngredients(cand.ingredients_raw);
    const belt = await matchDeterministic(parsed, phase, beltDomain);
    if (belt.dangerousMatch) continue;
    
    // ── NEW: Compute candidate's own GlowScore ──────────
    const candidateDangerCount = belt.matches.filter(m => m.riskLevel === 'danger').length;
    const candidateCautionCount = belt.matches.filter(m => m.riskLevel === 'caution').length;
    const candidateUnknownCount = belt.matches.filter(m => !m.matched).length;
    const candidateGlowScore = Math.max(0, Math.min(100,
      100 - (candidateDangerCount * 50) - (candidateCautionCount * 15) - (candidateUnknownCount * 2)
    ));
    
    // Get origin's GlowScore (stored by /api/scan in phaseCache)
    const originGlowScore = (phaseCache as any)?.glow_score ?? 50;
    
    // REJECT if not at least 10 points better
    if (candidateGlowScore < originGlowScore + 10) {
      req.log.info({
        candidateBarcode: b, candidateGlowScore, originGlowScore,
      }, 'alternative rejected: insufficient improvement');
      continue;
    }
    
    // Store computed score for DTO
    (cand as any)._glowScore = candidateGlowScore;
    
    // ── Existing categoryTag check (keep as is) ─────────
    if (categoryTag && liveStrategy === 'keywords' 
        && !cand.categories_tags.includes(categoryTag)) {
      continue;
    }
    
    // ── IMPROVED junk brand filter ──────────────────────
    // Reject products where brand equals category (e.g. brand: "Moutarde")
    // OR products without identifiable brand
    const brandNorm = cand.brand.trim().toLowerCase();
    const categoryNorm = cand.category.trim().toLowerCase();
    
    if (brandNorm.length < 2) {
      // No brand at all → reject unless image AND meaningful difference
      if (!cand.image_url) continue;
      cand.brand = 'Marque distributeur';
    } else if (brandNorm === categoryNorm) {
      // Brand = category name (OFF junk) → relabel
      cand.brand = 'Marque distributeur';
    }
    
    safeCandidates.push(cand);
  }
