Problème : quand l'OCR renvoie des ingrédients pourris ("saveur menthet vert",

affichés bruts. Casse la confiance.

ÉTAPE 2.1 — Créer un helper réutilisable

Créer le fichier : lib/ocrQuality.ts

  /**
   * Détecte si une liste OCR est de mauvaise qualité.
   * Retourne true si > 40% des lignes ont des problèmes détectables.
   */
  export function isLowQualityOcr(lines: string[] | undefined | null): boolean {
    if (!lines || lines.length === 0) return true;

    const issues = lines.filter((rawLine) => {
      const l = (rawLine ?? '').trim();
      // Trop court
      if (l.length < 3) return true;
      // Ligne qui se termine par % ou parenthèse non fermée
      if (/[%()][a-z]?\s*$/i.test(l)) return true;
      // Ligne qui se termine par ":" suivi de rien
      if (/:\s*$/.test(l)) return true;
      // Suite de 5+ consonnes (mot inventé probable)
      if (/[bcdfghjklmnpqrstvwxz]{5,}/i.test(l)) return true;
      // Caractères spéciaux suspects en chaîne
      if (/[#@~`^]{2,}/.test(l)) return true;
      return false;
    });

    return issues.length / lines.length > 0.4;
  }

ÉTAPE 2.2 — Tester avec un fichier test

Créer __tests__/ocrQuality.test.ts :

  import { isLowQualityOcr } from '@/lib/ocrQuality';

  describe('isLowQualityOcr', () => {
    test('detect bouillie évidente', () => {
      expect(isLowQualityOcr([
        'saveur menthet vert',
        'matieres grasses %dent',
        'placides:',
        'acides gras saturés: 0)',
      ])).toBe(true);
    });

    test('accepte liste propre', () => {
      expect(isLowQualityOcr([
        'eau',
        'sucre',
        'lait écrémé en poudre',
        'arôme naturel vanille',
        'sel',
      ])).toBe(false);
    });

    test('liste vide ou null = low quality', () => {
      expect(isLowQualityOcr([])).toBe(true);
      expect(isLowQualityOcr(null)).toBe(true);
      expect(isLowQualityOcr(undefined)).toBe(true);
    });

    test('liste mixte ≤ 40% problèmes = OK', () => {
      expect(isLowQualityOcr([
        'eau',
        'sucre',
        'lait',
        'arôme',
        'menthet:', // 1/5 = 20%, on accepte
      ])).toBe(false);
    });
  });
