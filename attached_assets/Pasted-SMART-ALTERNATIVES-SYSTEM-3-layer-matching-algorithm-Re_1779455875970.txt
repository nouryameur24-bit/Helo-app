export async function getAlternativesByBarcode(
  barcode: string,
  flaggedIngredientNames: string[],
  trimester: Trimester,
): Promise<AlternativeProduct[]> {
  if (!isSupabaseConfigured || !barcode) return [];
  
  const { data: current } = await supabase
    .from('products')
    .select('id, name, brand, category, ingredients_raw')
    .eq('barcode', barcode)
    .maybeSingle();
  
  if (!current) return [];
  
  // Extract product type keywords from the name
  const productKeywords = extractProductKeywords(current.name);
  // e.g. "Crème visage Nivea Q10" → ['crème', 'visage']
  
  const flaggedLower = flaggedIngredientNames.map(n => n.toLowerCase());
  
  // LAYER 1: Match by keywords in name
  if (productKeywords.length > 0) {
    let query = supabase
      .from('products')
      .select('id, name, brand, category, barcode, image_url, ingredients_raw')
      .eq('category', current.category)
      .neq('id', current.id)
      .not('ingredients_raw', 'is', null);
    
    // Build OR filter for each keyword
    const orFilter = productKeywords
      .map(kw => `name.ilike.%${kw}%`)
      .join(',');
    query = query.or(orFilter).limit(20);
    
    const { data: candidates } = await query;
    const safe = filterSafeProducts(candidates || [], flaggedLower);
    if (safe.length >= 3) return safe.slice(0, 3);
    
    // LAYER 2: Same brand
    if (safe.length < 3 && current.brand) {
      const { data: brandProducts } = await supabase
        .from('products')
        .select('id, name, brand, category, barcode, image_url, ingredients_raw')
        .eq('brand', current.brand)
        .neq('id', current.id)
        .not('ingredients_raw', 'is', null)
        .limit(20);
      
      const brandSafe = filterSafeProducts(brandProducts || [], flaggedLower);
      const combined = [...safe, ...brandSafe.filter(b => !safe.find(s => s.id === b.id))];
      if (combined.length >= 3) return combined.slice(0, 3);
      safe.push(...brandSafe);
    }
    
    if (safe.length > 0) return safe.slice(0, 3);
  }
  
  // LAYER 3: Hardcoded safe products for common categories
  const fallback = getHardcodedAlternatives(current.name, current.category);
  if (fallback.length > 0) return fallback;
  
  // Last resort: any product in same category, no flagged ingredients
  const { data: anyCat } = await supabase
    .from('products')
    .select('id, name, brand, category, barcode, image_url, ingredients_raw')
    .eq('category', current.category)
    .neq('id', current.id)
    .not('ingredients_raw', 'is', null)
    .limit(50);
  
  return filterSafeProducts(anyCat || [], flaggedLower).slice(0, 3);
}

// HELPERS

function extractProductKeywords(name: string): string[] {
  const lower = name.toLowerCase();
  const PRODUCT_TYPES = [
    'crème visage', 'crème mains', 'crème corps', 'crème solaire',
    'lait corps', 'lait visage', 'shampoing', 'après-shampoing',
    'gel douche', 'savon', 'déodorant', 'parfum', 'eau micellaire',
    'dentifrice', 'bain de bouche', 'fond de teint', 'mascara',
    'rouge à lèvres', 'baume lèvres', 'huile démaquillante',
    'sérum', 'masque visage', 'gommage', 'lotion tonique',
    'huile végétale', 'beurre karité', 'liniment', 'eau florale',
    'yaourt', 'fromage', 'jambon', 'saumon', 'thon', 'chocolat',
    'biscuit', 'pâte à tartiner', 'jus', 'eau', 'lait', 'beurre',
  ];
  
  return PRODUCT_TYPES.filter(type => lower.includes(type));
}

function filterSafeProducts(
  candidates: any[],
  flaggedLower: string[]
): AlternativeProduct[] {
  return candidates
    .filter(c => {
      const ingredientsLower = (c.ingredients_raw || '').toLowerCase();
      return !flaggedLower.some(flag => ingredientsLower.includes(flag));
    })
    .map(p => ({
      id: p.id,
      name: p.name,
      brand: p.brand,
      category: p.category,
      barcode: p.barcode,
      image_url: p.image_url,
      description_fr: null,
      overall_risk: 'safe',
      price_range: '',
      popularity_count: 0,
    }));
}

function getHardcodedAlternatives(
  productName: string,
  category: string
): AlternativeProduct[] {
  const lower = productName.toLowerCase();
  
  for (const [type, brands] of Object.entries(HARDCODED_ALTERNATIVES)) {
    if (lower.includes(type)) {
      // Try to find these branded products in Supabase
      // For now, return them as text-only suggestions
      return brands.map((brand, idx) => ({
        id: `fallback-${idx}`,
        name: brand,
        brand: brand.split(' ')[0],
        category: category as any,
        barcode: null,
        image_url: null,
        description_fr: 'Alternative recommandée pendant la grossesse',
        overall_risk: 'safe',
        price_range: '',
        popularity_count: 0,
      }));
    }
  }
  
  return [];
}
